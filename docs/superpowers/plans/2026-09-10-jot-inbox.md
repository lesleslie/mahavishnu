**Goal:** Build the zero-turn quick-capture inbox specified at `docs/superpowers/specs/2026-09-09-jot-inbox-design.md` — a `UserPromptSubmit` hook writes jots to an append-only log, a drain replicates to Dhara, Session-Buddy indexes from Dhara, and Mahavishnu MCP tools surface `/jot` review.

**Architecture:** Linear pipeline. Hook → `~/.mahavishnu/jot/log.jsonl` (durability) → drain (urllib, 2s) → Dhara :8683 (canonical) → Session-Buddy :8678 (derived index) → MCP tools (read local fold, not Dhara). **Tech Stack:** Python 3.14 stdlib (capture hook + codec), `urllib` (sync drain), `httpx2` (async drain), `mcp_common.fastmcp` (MCP tools), existing `mahavishnu` package patterns.

**Spec:** `docs/superpowers/specs/2026-09-09-jot-inbox-design.md` (commits `244b7cd1` → `95819359` → `ad620124`). This plan implements that spec; the spec travels with it.

## Global Constraints

- **Python 3.14 target.** `from __future__ import annotations` first non-comment line of every source file.
- **Line length 100.** `max-args 10`, `max-branches 15`, `max-returns 6`, `max-statements 55` (target 30).
- **Coverage 89%** (`--cov-fail-under=89.01682905225863` per `pyproject.toml`).
- **Stdlib only on capture path** (the hook, the vendored codec copy). No `oneiric.logging`, no `logging`, no `print` in the hook.
- **No `assert` in `mahavishnu/**`.** Use the `mahavishnu.core.errors` exception hierarchy.
- **Stdlib `json.dumps` only** for event serialization — not msgspec, not orjson, not pickle. Vendored copy in the hook uses the same options (`sort_keys=True, separators=(",", ":")`).

- **Three ops only: `add`, `done`, `reopen`.** No `redact` op (operator procedure handles removal; see spec §"Removal (no code)").
- **Stable handle: 6 hex chars** (the `0.003%` "per-insertion" figure is wrong; lifetime collision at 4 chars is ~70% by 400 jots).
- **Log mode `0o600`, dir `0o700`.** Set explicitly; never inherited from umask.
- **Markers:** `unit`, `integration`, `property`, `slow`. Per-test timeout 300 s; > 10 s tests are `@pytest.mark.slow`.

```python
"""Event codec for jot inbox. Stdlib json.dumps; one canonical function."""
from __future__ import annotations

import json
from typing import Any

EVENT_KEYS: frozenset[str] = frozenset({"op", "id", "hlc", "text", "ctx"})

# Bidi and zero-width characters that must not appear in stored text.
# From session-buddy/session_buddy/ingesters/redaction.py defense list.
_BIDI_AND_ZW: tuple[str, ...] = (
    "‎", "‏",  # LRM, RLM
    "‪", "‫", "‬", "‭", "‮",  # LRE, RLE, PDF, LRO, RLO
    "⁦", "⁧", "⁨", "⁩",  # LRI, RLI, FSI, PDI
)
_CONTROL_NON_NL: tuple[int, ...] = tuple(range(0, 32))  # U+0000-U+001F


def _sanitize_text(value: str) -> str:
    """Strip control chars (except \\n) and bidi/zero-width."""
    out = value.translate({cp: None for cp in _CONTROL_NON_NL if cp != ord("\n")})
    for ch in _BIDI_AND_ZW:
        out = out.replace(ch, "")
    return out


def _strip_for_keys(d: dict[str, Any]) -> dict[str, Any]:
    """Strip bidi/zero-width/control chars from all string values, recursively."""
    out: dict[str, Any] = {}
    for k, v in d.items():
        if isinstance(v, str):
            out[k] = _sanitize_text(v)
        elif isinstance(v, dict):
            out[k] = _strip_for_keys(v)
        else:
            out[k] = v
    return out


def serialize_event(event: dict[str, Any]) -> bytes:
    """Serialize a jot event to canonical bytes (single newline-terminated line).

    Invariant: identical input → identical output (sort_keys, no whitespace,
    trailing newline). The vendored copy in the hook must produce identical bytes.
    """
    sanitized = _strip_for_keys(event)
    line = json.dumps(
        sanitized, sort_keys=True, separators=(",", ":"), ensure_ascii=False
    ).encode("utf-8")
    return line + b"\n"


def deserialize_event(line: bytes) -> dict[str, Any]:
    """Strict deserializer. Raises ValueError on unknown top-level keys."""
    obj = json.loads(line)
    if not isinstance(obj, dict):
        raise ValueError(f"event must be object, got {type(obj).__name__}")
    extra = set(obj) - EVENT_KEYS
    if extra:
        raise ValueError(f"unknown event keys: {sorted(extra)}")
    missing = {"op", "id", "hlc"} - set(obj)
    if missing:
        raise ValueError(f"missing required event keys: {sorted(missing)}")
    return obj
```

```python
"""Stable handle resolution. Last 6 hex chars (or any unambiguous substring)."""
from __future__ import annotations


class AmbiguousJotId(ValueError):
    """Raised when a short ID query matches zero or multiple jots."""

    def __init__(self, query: str, candidates: list[str]) -> None:
        self.query = query
        self.candidates = candidates
        super().__init__(
            f"Short ID {query!r} ambiguous or unknown; "
            f"candidates: {candidates!r}"
        )


def _last_n(s: str, n: int) -> str:
    return s[-n:] if len(s) >= n else s


def resolve_short_id(fold: dict, query: str) -> str:
    """Resolve a short ID query to a full jot ID.

    Matching priority:
    1. Exact match (full ID)
    2. Last-6-char match
    3. Any unambiguous substring match (case-insensitive)
    """
    if not query:
        raise AmbiguousJotId(query, [])

    # 1. Exact match
    if query in fold:
        return query

    # 2. Last-6-char match
    q_lower = query.lower()
    by_suffix = [jid for jid in fold if _last_n(jid, 6).lower() == q_lower]
    if len(by_suffix) == 1:
        return by_suffix[0]
    if len(by_suffix) > 1:
        # Fall through to substring for diagnostic; ambiguous.
        pass

    # 3. Any unambiguous substring
    substr_matches = [jid for jid in fold if q_lower in jid.lower()]
    if len(substr_matches) == 1:
        return substr_matches[0]
    raise AmbiguousJotId(query, substr_matches or by_suffix)
```

```python
"""Ambient context types and validation for jot capture."""
from __future__ import annotations

import os
import re
from dataclasses import dataclass

BRANCH_MAX_BYTES: int = 1024
BRANCH_REGEX: re.Pattern[str] = re.compile(r"^[A-Za-z0-9._/-]+$")
_CONTROL_NON_NL: tuple[int, ...] = tuple(range(0, 32))


@dataclass(frozen=True)
class AmbientContext:
    repo: str | None
    branch: str | None
    sha: str | None
    cwd: str
    project_dir: str
    session_id: str

    def to_dict(self) -> dict:
        return {
            "repo": self.repo, "branch": self.branch, "sha": self.sha,
            "cwd": self.cwd, "project_dir": self.project_dir,
            "session_id": self.session_id,
        }


def _strip_control(s: str) -> str:
    return s.translate({cp: None for cp in _CONTROL_NON_NL if cp != ord("\n")})


def _realpath_absolute(path: str, *, field: str) -> str:
    if not path:
        raise ValueError(f"{field}: empty path")
    real = os.path.realpath(path)
    if not os.path.isabs(real):
        raise ValueError(f"{field}: not absolute after canonicalization: {path!r}")
    return real


def _resolve_repo_from_worktree(project_dir: str) -> str | None:
    """If project_dir is a worktree, return the canonical repo path."""
    # Heuristic: ends with /<name>/.git/worktrees/<name>; canonical is two
    # parents up. We refuse to walk git internals (per spec).
    marker = "/.git/worktrees/"
    idx = project_dir.find(marker)
    if idx == -1:
        return None
    head = project_dir[:idx]
    return head or None


def validate_ambient(raw: dict) -> AmbientContext:
    """Validate and canonicalize an ambient context dict from the hook."""
    branch_raw = raw.get("branch")
    branch: str | None
    if branch_raw is None:
        branch = None
    else:
        branch = _strip_control(branch_raw)
        if ".." in branch:
            raise ValueError(f"branch contains '..': {branch!r}")
        if len(branch.encode("utf-8")) > BRANCH_MAX_BYTES:
            raise ValueError(f"branch too long (> {BRANCH_MAX_BYTES} bytes)")
        if not BRANCH_REGEX.match(branch):
            raise ValueError(f"branch does not match {BRANCH_REGEX.pattern!r}: {branch!r}")

    cwd = _realpath_absolute(raw["cwd"], field="cwd")
    project_dir = _realpath_absolute(raw["project_dir"], field="project_dir")
    repo = _resolve_repo_from_worktree(project_dir) or project_dir

    session_id = _strip_control(raw["session_id"])
    if not session_id:
        raise ValueError("session_id: empty")

    return AmbientContext(
        repo=repo, branch=branch, sha=raw.get("sha"),
        cwd=cwd, project_dir=project_dir, session_id=session_id,
    )
```

```python
"""Two-pass fold: pass one applies `add`; pass two applies `done`/`reopen`.

Ops referencing unknown IDs in pass two are parked and re-applied. HLC
ordering resolves cross-machine divergence.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

OPEN: str = "OPEN"
DONE: str = "DONE"


@dataclass(frozen=True, order=True)
class HLC:
    wall: int
    ctr: int
    node: str

    # `order=True` on the dataclass generates __lt__, __le__, __gt__, __ge__,
    # __eq__ from the field order. fold() uses >=, so this MUST be present.

    @classmethod
    def from_dict(cls, d: dict[str, Any]) -> "HLC":
        return cls(wall=d["wall"], ctr=d["ctr"], node=d["node"])


@dataclass(frozen=True)
class FoldedJot:
    id: str
    op: str  # last op applied
    text: str
    ctx: dict[str, Any]
    hlc: HLC
    status: str


@dataclass(frozen=True)
class FoldResult:
    jots: dict[str, FoldedJot]
    malformed_count: int
    parked_count: int


def _status_for(op: str) -> str:
    return DONE if op == "done" else OPEN


def fold(events: list[dict[str, Any]]) -> FoldResult:
    """Apply ops in two passes with HLC ordering."""
    state: dict[str, FoldedJot] = {}
    parked: list[dict[str, Any]] = []
    malformed = 0

    def apply(ev: dict[str, Any]) -> bool:
        nonlocal malformed
        try:
            op = ev["op"]
            jid = ev["id"]
            hlc = HLC.from_dict(ev["hlc"])
        except (KeyError, TypeError):
            malformed += 1
            return False
        if op == "add":
            state[jid] = FoldedJot(
                id=jid, op=op, text=ev.get("text", ""), ctx=ev.get("ctx", {}),
                hlc=hlc, status=OPEN,
            )
        elif op == "done":
            if jid in state:
                prev = state[jid]
                if hlc >= prev.hlc:
                    state[jid] = FoldedJot(
                        id=jid, op=op, text=prev.text, ctx=prev.ctx,
                        hlc=hlc, status=DONE,
                    )
            else:
                return False  # park
        elif op == "reopen":
            if jid in state:
                prev = state[jid]
                if hlc >= prev.hlc:
                    state[jid] = FoldedJot(
                        id=jid, op=op, text=prev.text, ctx=prev.ctx,
                        hlc=hlc, status=OPEN,
                    )
            else:
                return False
        else:
            malformed += 1
            return False
        return True

    # Pass 1: adds (in HLC order)
    adds = sorted(
        (e for e in events if e.get("op") == "add"),
        key=lambda e: HLC.from_dict(e["hlc"]),
    )
    for ev in adds:
        apply(ev)

    # Pass 2: mutations (in HLC order), with parking
    mutations = sorted(
        (e for e in events if e.get("op") in ("done", "reopen")),
        key=lambda e: HLC.from_dict(e["hlc"]),
    )
    for ev in mutations:
        if not apply(ev):
            parked.append(ev)

    # Replay parked (handles cross-machine where add arrives after)
    for ev in parked:
        apply(ev)

    return FoldResult(
        jots=state, malformed_count=malformed, parked_count=len(parked),
    )
```

```python
"""Offset bookkeeping with inode+size integrity guard."""
from __future__ import annotations

import json
import os
import tempfile
from dataclasses import asdict, dataclass
from pathlib import Path

from mahavishnu.jot.paths import offset_path


@dataclass(frozen=True)
class OffsetState:
    byte_offset: int
    log_inode: int
    log_size: int


class OffsetMismatch(ValueError):
    """Raised when the offset file's inode/size no longer matches the log."""


def read_offset() -> OffsetState:
    path = offset_path()
    if not path.exists():
        return OffsetState(0, 0, 0)
    data = json.loads(path.read_text())
    return OffsetState(
        byte_offset=int(data.get("byte_offset", 0)),
        log_inode=int(data.get("log_inode", 0)),
        log_size=int(data.get("log_size", 0)),
    )


def write_offset(state: OffsetState) -> None:
    path = offset_path()
    path.parent.mkdir(parents=True, exist_ok=True)
    os.chmod(path.parent, 0o700)
    # Atomic write: temp + rename
    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=".offset.")
    try:
        with os.fdopen(fd, "w") as f:
            json.dump(asdict(state), f, sort_keys=True)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
    finally:
        if os.path.exists(tmp):
            os.unlink(tmp)


def reset_offset() -> None:
    """Zero the offset field only; preserve inode/size for the integrity guard."""
    cur = read_offset()
    write_offset(OffsetState(byte_offset=0, log_inode=cur.log_inode, log_size=cur.log_size))


def verify_against_log(log: Path) -> None:
    """Raise OffsetMismatch if the offset's inode/size don't match the log."""
    state = read_offset()
    if not log.exists():
        if state.byte_offset == 0:
            return
        raise OffsetMismatch("log missing but offset is non-zero")
    st = os.stat(log)
    if st.st_ino != state.log_inode:
        raise OffsetMismatch(
            f"inode mismatch: offset says {state.log_inode}, log has {st.st_ino}"
        )
    if st.st_size < state.log_size:
        # Log shrunk (e.g., backup restore) — must re-drain
        raise OffsetMismatch(
            f"log shrunk: offset says {state.log_size}, log has {st.st_size}"
        )
```

```python
"""Pure planner: events + offset → replication plan. No I/O."""
from __future__ import annotations

from dataclasses import dataclass

BATCH_MAX_EVENTS: int = 100


@dataclass(frozen=True)
class EventBatch:
    events: list[bytes]
    last_byte: int  # offset after the last complete line in this batch


@dataclass(frozen=True)
class ReplicationPlan:
    batches: list[EventBatch]


def _find_last_newline(buffer: bytes) -> int:
    """Return the byte index AFTER the last newline, or 0 if none."""
    idx = buffer.rfind(b"\n")
    return idx + 1 if idx >= 0 else 0


def plan_replication(events: list[bytes], current_offset: int) -> ReplicationPlan:
    """Slice events into batches, never reading past the last complete line."""
    boundary = _find_last_newline(b"".join(events))
    if boundary == 0:
        return ReplicationPlan(batches=[])
    complete = b"".join(events)[:boundary]
    lines = [ln for ln in complete.split(b"\n") if ln]
    batches: list[EventBatch] = []
    for i in range(0, len(lines), BATCH_MAX_EVENTS):
        chunk = lines[i:i + BATCH_MAX_EVENTS]
        # Reassemble with newlines for byte accounting
        joined = b"\n".join(chunk) + b"\n"
        last_byte = current_offset + len(joined)
        batches.append(EventBatch(events=chunk, last_byte=last_byte))
    return ReplicationPlan(batches=batches)
```

```python
"""Sync drain (urllib, 2 s hard limit). Used by SessionStart, SessionEnd,
and drain-before-read on every MCP tool call.
"""
from __future__ import annotations

import json
import os
import socket
import urllib.error
import urllib.request
from dataclasses import dataclass

from mahavishnu.jot.drain_core import plan_replication
from mahavishnu.jot.events import deserialize_event
from mahavishnu.jot.offsets import (
    OffsetState, read_offset, verify_against_log, write_offset, reset_offset,
)
from mahavishnu.jot.paths import lock_path, log_path

# redact is imported lazily inside _redact_event() so that drain_sync can
# be loaded even if session-buddy is not yet importable (test isolation
# and forward-compat with the drain-time redaction module).


@dataclass(frozen=True)
class DrainResult:
    replicated_count: int
    advanced_to: int
    dhara_503: bool
    timed_out: bool


# URL allowlist for the drain — prevents a compromised MCP caller from
# redirecting the log to an attacker server. Configured via
# ``MAHAVISHNU_DHARA_URL`` env var; defaults to localhost.
_DHARA_URL_ENV: str = "MAHAVISHNU_DHARA_URL"
_DEFAULT_DHARA_URL: str = "http://localhost:8683"
_ALLOWED_SCHEMES: tuple[str, ...] = ("http", "https")
_LOCALHOST_HOSTS: frozenset[str] = frozenset({"localhost", "127.0.0.1", "::1"})


def _resolve_drain_url() -> str:
    """Return the configured drain URL (env var or default)."""
    return os.environ.get(_DHARA_URL_ENV, _DEFAULT_DHARA_URL)


def _validate_drain_url(url: str) -> str:
    """Validate the drain URL against the allowlist. Raises ValueError
    on a disallowed scheme or host. The allowlist is intentionally
    strict: drain must go to the configured Dhara instance, not anywhere
    a caller chooses. Operators who need a non-localhost Dhara must set
    ``MAHAVISHNU_DHARA_URL`` explicitly; that host is then allowed.
    """
    from urllib.parse import urlparse
    parsed = urlparse(url)
    if parsed.scheme not in _ALLOWED_SCHEMES:
        raise ValueError(
            f"drain url scheme {parsed.scheme!r} not in {_ALLOWED_SCHEMES}"
        )
    host = parsed.hostname or ""
    allowed = _LOCALHOST_HOSTS | {_resolve_drain_url().split("//", 1)[-1].split(":", 1)[0]}
    if host not in allowed:
        raise ValueError(
            f"drain url host {host!r} is not in the allowlist; "
            f"set {_DHARA_URL_ENV} to an allowed host"
        )
    return url


def _redact_event(event: dict) -> dict:
    """Lazy import of session-buddy redact so drain_sync can be loaded in
    environments where session-buddy isn't installed (e.g., unit tests
    that don't exercise redaction)."""
    from mahavishnu.jot.redact import redact_event
    return redact_event(event)


def _read_log_bytes() -> list[bytes]:
    p = log_path()
    if not p.exists():
        return []
    return p.read_bytes().split(b"\n")


def _post_batch(url: str, body: bytes, timeout: float) -> tuple[bool, bool]:
    """Post a batch. Returns (ok, timed_out). 5xx treated as failure but not timeout."""
    req = urllib.request.Request(
        url, data=body, method="POST",
        headers={"Content-Type": "application/json"},
    )
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return (200 <= resp.status < 300, False)
    except socket.timeout:
        return (False, True)
    except urllib.error.HTTPError as e:
        return (e.code < 500, False)  # 4xx is terminal; 5xx is retryable
    except (urllib.error.URLError, ConnectionError, OSError):
        return (False, False)


def _acquire_lock() -> int | None:
    """Best-effort non-blocking lock. Returns fd or None."""
    p = lock_path()
    try:
        fd = os.open(str(p), os.O_CREAT | os.O_RDWR | os.O_EXCL | os.O_NOFOLLOW, 0o600)
        os.write(fd, str(os.getpid()).encode())
        return fd
    except FileExistsError:
        return None


def _release_lock(fd: int | None) -> None:
    if fd is not None:
        try:
            os.close(fd)
        finally:
            try:
                os.unlink(str(lock_path()))
            except FileNotFoundError:
                pass


def drain_sync(
    base_url: str = "http://localhost:8683",
    timeout_seconds: float = 2.0,
) -> DrainResult:
    """Drain pending events to Dhara. Returns DrainResult; on any failure
    leaves the offset unchanged so the next trigger retries."""
    base_url = _validate_drain_url(base_url)
    fd = _acquire_lock()
    if fd is None:
        return DrainResult(0, read_offset().byte_offset, False, False)

    try:
        log = log_path()
        try:
            verify_against_log(log)
        except Exception:
            # Mismatch (home-dir sync, restore): reset and continue
            reset_offset()

        offset = read_offset()
        all_lines = _read_log_bytes()
        # Skip lines we've already replicated
        if offset.byte_offset > 0:
            skipped = 0
            new_lines: list[bytes] = []
            for ln in all_lines:
                if skipped < offset.byte_offset:
                    skipped += len(ln) + 1  # +1 for newline
                    continue
                new_lines.append(ln)
            lines = new_lines
        else:
            lines = all_lines

        plan = plan_replication(lines, current_offset=offset.byte_offset)
        replicated = 0
        advanced = offset.byte_offset
        timed_out = False
        saw_503 = False

        for batch in plan.batches:
            # Redact each event before sending
            redacted: list[bytes] = []
            for line in batch.events:
                try:
                    ev = deserialize_event(line)
                    redacted.append(json.dumps(_redact_event(ev)).encode("utf-8") + b"\n")
                except ValueError:
                    continue  # malformed; skip
            if not redacted:
                continue
            body = b"[" + b",".join(redacted) + b"]"
            ok, to = _post_batch(f"{base_url}/jot/replicate", body, timeout_seconds)
            if to:
                timed_out = True
                break
            if not ok:
                saw_503 = True
                break
            replicated += len(redacted)
            advanced = batch.last_byte

        if replicated > 0:
            log_st = log.stat()
            write_offset(OffsetState(
                byte_offset=advanced,
                log_inode=log_st.st_ino,
                log_size=log_st.st_size,
            ))

        return DrainResult(
            replicated_count=replicated, advanced_to=advanced,
            dhara_503=saw_503, timed_out=timed_out,
        )
    finally:
        _release_lock(fd)
```

```python
import sys
from dataclasses import dataclass
from typing import Iterable

from mahavishnu.jot.fold import FoldedJot, OPEN, DONE

_STATUS_GLYPHS: dict[str, str] = {OPEN: "OPEN", DONE: "DONE ", "REOPEN": "REOP "}

# Truncation glyph
ELLIPSIS: str = "…"


@dataclass(frozen=True)
class Vitals:
    open_count: int
    oldest_days: int | None
    captures_7d: int
    reviews_7d: int
    threshold_glyph: str

    @property
    def capture_review_ratio(self) -> float:
        if self.reviews_7d == 0:
            return float("inf") if self.captures_7d > 0 else 0.0
        return self.captures_7d / self.reviews_7d


def compute_vitals(
    *, open_count: int, oldest_days: int | None,
    captures_7d: int, reviews_7d: int,
) -> Vitals:
    ratio = (captures_7d / reviews_7d) if reviews_7d > 0 else (
        float("inf") if captures_7d > 0 else 0.0
    )
    if ratio > 10:
        glyph = "· ⚠⚠"
    elif ratio > 3:
        glyph = "· ⚠"
    else:
        glyph = ""
    return Vitals(
        open_count=open_count, oldest_days=oldest_days,
        captures_7d=captures_7d, reviews_7d=reviews_7d,
        threshold_glyph=glyph,
    )


def render_vitals(v: Vitals) -> str:
    parts = [f"{v.open_count} open"]
    if v.oldest_days is not None:
        parts.append(f"oldest {v.oldest_days}d")
    if v.threshold_glyph:
        parts.append(v.threshold_glyph)
    return " · ".join(parts)

def _format_age(seconds_since_capture: int) -> str:
    if seconds_since_capture < 60:
        return f"{seconds_since_capture}s"
    if seconds_since_capture < 3600:
        return f"{seconds_since_capture // 60}m"
    if seconds_since_capture < 86400:
        return f"{seconds_since_capture // 3600}h"
    return f"{seconds_since_capture // 86400}d"


def _format_seen(n: int) -> str:
    if n <= 1:
        return ""
    if n == 2:
        return "2nd time"
    if n == 3:
        return "3rd time"
    return f"{n}×"  # n×

MIN_TEXT_WIDTH: int = 8


def _is_tty() -> bool:
    if not sys.stdout.isatty():
        return False
    if os.environ.get("NO_COLOR"):
        return False
    if os.environ.get("TERM") == "dumb":
        return False
    if os.environ.get("CI"):
        return False
    return True


def _truncate(text: str, width: int) -> str:
    if width <= 0:
        return ""
    if len(text) <= width:
        return text
    if width <= 1:
        return ELLIPSIS
    return text[: width - 1] + ELLIPSIS


def _short_id(jid: str) -> str:
    return jid[-6:]


def render_list_row(
    jot: FoldedJot, *, terminal_width: int = 120, is_tty: bool = True,
    seen_count: int = 1,
) -> str:
    """Render one jot as a list row. Space-aligned, no bullets in rows."""
    if not is_tty:
        terminal_width = 80
    status = _STATUS_GLYPHS.get(jot.status, jot.status[:4].ljust(4))
    age = _format_age(0)  # caller computes from capture time
    # Priority drop: seen > age > text
    cols = [f"{_short_id(jot.id)}", status, age]
    if seen_count > 1:
        cols.append(_format_seen(seen_count))
    # Remaining width for text
    used = sum(len(c) + 1 for c in cols)  # +1 per gap
    text_w = max(MIN_TEXT_WIDTH, terminal_width - used)
    text = _truncate(jot.text, text_w)
    return " ".join(cols + [text])


def render_show(jot: FoldedJot) -> dict:
    """Render a single jot for /jot show N. Per spec: 3-section fixed format.

    Returns a dict (the MCP tool wraps it; the CLI prints it). The three
    sections:
    - Line 1: <6-char-id> <4-char-status> <age>  (always present)
    - Line 2: repo:<repo> branch:<branch> sha:<7-char-sha>
    - Lines 3+: full text, never truncated (newlines escaped as \\n)

    repo = ctx.get("repo", "?")
    branch = ctx.get("branch", "?")
    sha = (ctx.get("sha") or "?")[:7]
    return {
        "header": f"{_short_id(jot.id)} {_STATUS_GLYPHS.get(jot.status, '    ')} {age}",
        "context": f"repo:{repo} branch:{branch} sha:{sha}",
        "text": jot.text,  # NEVER truncate, per spec
    }
```

```python
"""UserPromptSubmit capture hook for the jot inbox. Stdlib only.

The hook never raises; it always returns an exit code. On any failure it
returns 0 (pass-through) so the prompt is not lost. On a successful capture
with `,,` prefix it returns 2 (block) and writes a confirmation echo to
stdout, which Claude Code surfaces to the user.

Per spec, this file is the only Python that runs on the capture path. It
must not import from ``mahavishnu.*``; it vendors its own codec copy
(``_vendored_codec.py``).
"""
from __future__ import annotations

import json
import os
import re
import sys
import time
import uuid
from pathlib import Path

# Local vendored codec (byte-identical to mahavishnu.jot.events)
from _vendored_codec import serialize_event

MAX_JOT_TEXT_CHARS: int = 300
PREFIX: str = ",,"


def _log_path() -> Path:
    override = os.environ.get("MAHAVISHNU_JOT_DIR")
    base = Path(override) if override else Path.home() / ".mahavishnu" / "jot"
    return base / "log.jsonl"


def _ensure_dir(p: Path) -> None:
    """Ensure the jot dir exists with mode 0o700. Fail-open on error."""
    try:
        p.mkdir(parents=True, exist_ok=True)
        os.chmod(p, 0o700)
    except OSError:
        raise


def _redact_text(s: str) -> tuple[str, int, list[str]]:
    """Return (redacted_text, count, types). Stdlib only.

    Vendored high-confidence subset of the patterns in
    session_buddy.ingesters.redaction.redact. The full set runs at drain
    time (Task 10); this subset covers the common cases the hook can
    see on the capture path.
    """
    patterns: list[tuple[str, str]] = [
        (r"AKIA[0-9A-Z]{16}", "AKIA"),
        (r"ghp_[A-Za-z0-9]{36}", "ghp_"),
        (r"eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}", "JWT"),
        (r"(?i)password\s*[=:]\s*\S+", "password="),
        (r"Authorization:\s*\S+", "Authorization:"),
    ]
    count = 0
    types: set[str] = set()
    out = s
    for pat, t in patterns:
        new_out, n = re.subn(pat, f"[REDACTED:{t}]", out)
        if n:
            count += n
            types.add(t)
            out = new_out
    return out, count, sorted(types)


def _read_stdin() -> dict:
    raw = sys.stdin.read()
    if not raw:
        return {}
    return json.loads(raw)


def _extract_text(prompt: str) -> str | None:
    """Return the captured text if prefix is present, else None.

    Also enforces MAX_JOT_TEXT_CHARS and the multi-line shape gate.
    Returns None for pass-through (no prefix, or oversized, or multi-line).
    """
    # Strip all leading whitespace (Claude often emits prompts with one
    # or more leading spaces, and the prefix should still work). The
    # strip is intentionally narrow so a prompt that starts with ",,"
    # is never mangled into a non-prefix — the actual prefix check is
    # against the stripped text.
    prompt = prompt.lstrip()
    if not prompt.startswith(PREFIX):
        return None
    text = prompt[len(PREFIX):].lstrip()
    if "\n" in text:
        return None
    if len(text) > MAX_JOT_TEXT_CHARS:
        return None
    if not text:
        return None
    return text


def _read_git_metadata() -> dict:
    """Read repo/branch/sha from .git/HEAD, walking up from cwd. No subprocess.

    Returns empty dict on any failure (spec says capture anyway).
    The branch is validated against the spec's BRANCH_REGEX (subset of
    git's ref rules) and capped at BRANCH_MAX_BYTES (1024) — same rules
    that ``mahavishnu.jot.ctx.validate_ambient`` enforces on the package
    side. The hook is stdlib-only and cannot import that helper, so the
    rules are replicated inline.
    """
    cwd = os.environ.get("CLAUDE_PROJECT_DIR") or os.getcwd()
    cur = Path(cwd)
    for _ in range(8):
        head = cur / ".git" / "HEAD"
        if head.is_file():
            try:
                content = head.read_text().strip()
                if content.startswith("ref:"):
                    ref_path = cur / ".git" / content.split(":", 1)[1].strip()
                    if ref_path.is_file():
                        sha = ref_path.read_text().strip()[:7]
                    else:
                        sha = None
                    branch = content.split("/")[-1]
                    # Inline validation (replicates validate_ambient).
                    if ".." in branch or len(branch.encode("utf-8")) > 1024:
                        return {"repo": str(cur), "branch": None, "sha": sha}
                    if not re.match(r"^[A-Za-z0-9._/-]+$", branch):
                        return {"repo": str(cur), "branch": None, "sha": sha}
                    return {"repo": str(cur), "branch": branch, "sha": sha}
            except OSError:
                pass
            return {}
        parent = cur.parent
        if parent == cur:
            break
        cur = parent
    return {}


def _canonicalize_path(path: str) -> str | None:
    """realpath + absolute check, returning None on any failure. Inline
    version of ``mahavishnu.jot.ctx._realpath_absolute``."""
    if not path:
        return None
    try:
        real = os.path.realpath(path)
    except OSError:
        return None
    if not os.path.isabs(real):
        return None
    return real


def _build_event(text: str, ctx_extra: dict) -> dict:
    now_ms = int(time.time() * 1000)
    node = os.uname().nodename if hasattr(os, "uname") else "unknown"
    return {
        "op": "add",
        "id": str(uuid.uuid4()),
        "hlc": {"wall": now_ms, "ctr": 0, "node": node},
        "text": text,
        "ctx": {**ctx_extra, "session_id": ctx_extra.get("session_id", "unknown")},
    }


def _append_log(path: Path, line: bytes) -> bool:
    """Open log with O_CREAT|O_WRONLY|O_APPEND, mode 0o600, write line.

    Returns True on success, False on any failure (fail-open).
    A short write is a failure (spec: PIPE_BUF doesn't apply to regular files).
    """
    try:
        fd = os.open(
            str(path),
            os.O_CREAT | os.O_WRONLY | os.O_APPEND | os.O_NOFOLLOW,
            0o600,
        )
    except OSError:
        return False
    try:
        written = os.write(fd, line)
        return written == len(line)
    except OSError:
        return False
    finally:
        try:
            os.close(fd)
        except OSError:
            pass


def _echo(jot_id: str, text: str, red_count: int = 0, red_types: list | None = None) -> str:
    """Render the capture-time echo to stdout.

    ``text`` MUST be the redacted form (the implementer threads
    ``redacted_text`` from the capture-time redaction step). If
    ``red_count`` is non-zero, the echo prints the redaction summary
    instead of the text snippet — so secrets never appear in the echo.
    """
    if red_count > 0:
        types = ",".join(red_types or [])
        return f"jotted {jot_id[-6:]} · redacted:{red_count} ({types})"
    snippet = text if len(text) <= 30 else text[:29] + "…"
    return f"jotted {jot_id[-6:]} · \"{snippet}\""


def main() -> int:
    """Entry point. Returns 0 (pass-through) or 2 (block + echo)."""
    try:
        payload = _read_stdin()
        prompt = payload.get("prompt", "")
        text = _extract_text(prompt)
        if text is None:
            return 0  # no prefix, oversized, or multi-line: pass through

        # Capture-time redaction (inline subset; the full redact runs at
        # drain time). Both layers must produce the same text for the
        # echoed output.
        redacted_text, red_count, red_types = _redact_text(text)
        text_for_storage = redacted_text

        # Build event
        git = _read_git_metadata()
        # Canonicalize cwd / project_dir through the inline helper. The
        # hook cannot import validate_ambient (stdlib-only), so the same
        # realpath+absolute check is replicated. None on failure → still
        # capture, but the field is null (per spec).
        raw_cwd = os.getcwd()
        raw_project = os.environ.get("CLAUDE_PROJECT_DIR") or os.getcwd()
        ctx = {
            "repo": git.get("repo"),
            "branch": git.get("branch"),
            "sha": git.get("sha"),
            "cwd": _canonicalize_path(raw_cwd) or raw_cwd,
            "project_dir": _canonicalize_path(raw_project) or raw_project,
            "session_id": payload.get("session_id", "unknown"),
        }
        event = _build_event(text_for_storage, ctx)
        line = serialize_event(event)

        # Append
        log = _log_path()
        try:
            _ensure_dir(log.parent)
        except OSError:
            return 0
        if not _append_log(log, line):
            return 0  # fail-open: short write or other I/O failure

        # Block + echo. CRITICAL: echo uses redacted_text, not text —
        # secrets must never reach stdout (which Claude surfaces to the user).
        sys.stdout.write(_echo(event["id"], text_for_storage, red_count, red_types) + "\n")
        sys.stdout.flush()
        return 2
    except Exception:
        # Last-resort fail-open: any unexpected error → pass through
        return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def _child(barrier, log_path_str, round_idx, child_idx, tmp_str):
    barrier.wait()

    event = {
        "op": "add", "id": f"{round_idx:04d}{child_idx:04d}",

@pytest.mark.slow
def test_concurrent_append_no_interleaving(tmp_path):
    """Run 32 procs × 100 rounds, all waiting on a single Barrier per round.

    Earlier draft had 32 parties but spawned 3200 procs against the same
    barrier — every process past #32 raised BrokenBarrierError. The
    correct shape is: 32 procs, all waiting on a barrier of 32, repeated
    100 times.

        for c in range(n_procs):
            p = mp.Process(
                target=_child,
                args=(barrier, str(log), r, c, str(tmp_path)),
            )
            p.start()
            procs.append(p)

            p.join()

    # Read log and fold; every line must be a valid event
    lines = [ln for ln in log.read_bytes().split(b"\n") if ln]
    assert len(lines) == n_procs * n_rounds
    events = [deserialize_event(ln) for ln in lines]
    result = fold(events)
    assert result.malformed_count == 0
    assert len(result.jots) == n_procs * n_rounds
```

```python
from __future__ import annotations

from hypothesis import given, strategies as st

from mahavishnu.jot.fold import fold, OPEN, DONE


@given(
    wall_a=st.integers(min_value=0, max_value=10**12),
    wall_b=st.integers(min_value=0, max_value=10**12),
    ctr_a=st.integers(min_value=0, max_value=1000),
    ctr_b=st.integers(min_value=0, max_value=1000),
)
def test_hlc_orders_correctly_under_clock_skew(wall_a, wall_b, ctr_a, ctr_b):
    """Per spec: HLC (wall, ctr, node) ordering wins, not wall alone.

    Generate events with deliberately skewed wall values; assert the fold
    uses HLC tuple ordering (lexicographic over the full 3-tuple), not
    wall-clock last-write-wins. The node tiebreaker matters when wall
    AND ctr are equal — without it, Hypothesis finds a counter-example
    on the first run.
    """
    a = {"op": "add", "id": "x", "hlc": {"wall": wall_a, "ctr": ctr_a, "node": "n1"},
         "text": "a", "ctx": {"session_id": "s"}}
    b = {"op": "add", "id": "x", "hlc": {"wall": wall_b, "ctr": ctr_b, "node": "n2"},
         "text": "b", "ctx": {"session_id": "s"}}
    result = fold([a, b])
    j = result.jots["x"]
    # HLC lexicographic — node tiebreaker included.
    a_hlc = (wall_a, ctr_a, "n1")
    b_hlc = (wall_b, ctr_b, "n2")
    expected_text = "a" if a_hlc >= b_hlc else "b"
    assert j.text == expected_text


@given(seed=st.integers(min_value=0, max_value=10**6))
```

```python
"""Pure CLI logic for `mahavishnu jot ...`. Testable without Typer."""
from __future__ import annotations

import os
import sys
from typing import Sequence

from mahavishnu.jot.drain_sync import drain_sync
from mahavishnu.jot.events import serialize_event
from mahavishnu.jot.fold import fold, OPEN
from mahavishnu.jot.paths import log_path
from mahavishnu.jot.render import (
    render_list_row, render_show, render_echo, compute_vitals, Vitals,
)
from mahavishnu.jot.short_id import AmbiguousJotId, resolve_short_id


def _read_state() -> dict:
    p = log_path()
    if not p.exists():
        return {}
    import json
    events = []
    for ln in p.read_bytes().split(b"\n"):
        if not ln:
            continue
        try:
            events.append(json.loads(ln))
        except ValueError:
            continue
    return fold(events).jots


def run_jot_cli(args: Sequence[str]) -> int:
    if not args:
        return _cmd_list(["--all"])
    cmd, rest = args[0], list(args[1:])
    if cmd == "list":
        return _cmd_list(rest)
    if cmd == "show":
        return _cmd_show(rest)
    if cmd == "done":
        return _cmd_done(rest)
    if cmd == "reopen":
        return _cmd_reopen(rest)
    if cmd == "search":
        return _cmd_search(rest)
    if cmd == "relevant":
        return _cmd_relevant(rest)
    if cmd == "sync":
        return _cmd_sync(rest)
    print(f"unknown subcommand: {cmd}", file=sys.stderr)
    return 2


def _cmd_relevant(args: list[str]) -> int:
    """Local ranking: text overlap with the current session's repo,
    branch, recent files, and recent prompts. Per spec §Decision 6,
    v1 uses local ranking; v1.1 swaps in Session-Buddy embeddings.
    No Session-Buddy dependency in v1.
    """
    # Heuristic inputs — the actual session context would be plumbed
    # via env vars in a real implementation; for the spec compliance
    # we use simple text overlap.
    state = _read_state()
    cwd = os.getcwd()
    branch_hint = os.environ.get("JOT_RELEVANT_BRANCH", "")
    session_prompts: list[str] = os.environ.get("JOT_RELEVANT_PROMPTS", "").split("\n")
    hint_text = " ".join([cwd, branch_hint, *session_prompts]).lower()
    scored: list[tuple[float, object]] = []
    for j in state.values():
        if j.status != OPEN:
            continue
        # Token overlap
        hint_tokens = set(hint_text.split())
        jot_tokens = set(j.text.lower().split())
        score = len(hint_tokens & jot_tokens) / max(len(hint_tokens | jot_tokens), 1)
        scored.append((score, j))
    scored.sort(key=lambda x: x[0], reverse=True)
    for score, j in scored[:5]:
        print(render_list_row(j, terminal_width=120, is_tty=sys.stdout.isatty()))
    return 0


def _cmd_list(args: list[str]) -> int:
    state = _read_state()
    open_jots = [j for j in state.values() if j.status == OPEN]
    open_jots.sort(key=lambda x: x.hlc, reverse=True)
    if "--all" not in args:
        open_jots = open_jots[:5]
    for j in open_jots:
        print(render_list_row(j, terminal_width=120, is_tty=sys.stdout.isatty()))
    return 0


def _cmd_show(args: list[str]) -> int:
    if not args:
        print("usage: jot show <id>", file=sys.stderr)
        return 2
    state = _read_state()
    try:
        full_id = resolve_short_id(state, args[0])
    except AmbiguousJotId as e:
        print(str(e), file=sys.stderr)
        return 1
    print(render_show(state[full_id]))
    return 0


def _cmd_done(args: list[str]) -> int:
    return _apply_op("done", args)


def _cmd_reopen(args: list[str]) -> int:
    return _apply_op("reopen", args)


def _apply_op(op: str, args: list[str]) -> int:
    if not args:
        print(f"usage: jot {op} <id>", file=sys.stderr)
        return 2
    state = _read_state()
    try:
        full_id = resolve_short_id(state, args[0])
    except AmbiguousJotId as e:
        print(str(e), file=sys.stderr)
        return 1
    import time, uuid
    event = {
        "op": op, "id": full_id,
        "hlc": {"wall": int(time.time() * 1000), "ctr": 0, "node": os.uname().nodename},
    }
    line = serialize_event(event)
    p = log_path()
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open("ab") as f:
        f.write(line)
    return 0


def _cmd_search(args: list[str]) -> int:
    if not args:
        return 2
    state = _read_state()
    q = args[0].lower()
    matches = [j for j in state.values() if q in j.text.lower()]
    for j in matches[:20]:
        print(render_list_row(j, terminal_width=120, is_tty=sys.stdout.isatty()))
    return 0


def _cmd_sync(args: list[str]) -> int:
    result = drain_sync()
    print(f"replicated={result.replicated_count} advanced_to={result.advanced_to}")
    return 0
```
